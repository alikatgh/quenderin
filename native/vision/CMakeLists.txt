cmake_minimum_required(VERSION 3.22)
# May be included by Android/Apple builds which already own llama + ggml.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(quenderin_vision_runtime LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED PRISM_SOURCE_DIR)
  message(FATAL_ERROR "Set PRISM_SOURCE_DIR to the exact prepared Prism checkout")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/prepare_prism_runtime.py"
  verify --source "${PRISM_SOURCE_DIR}" RESULT_VARIABLE VERIFIED OUTPUT_VARIABLE PROVENANCE ERROR_VARIABLE VERIFY_ERROR)
if(NOT VERIFIED EQUAL 0)
  message(FATAL_ERROR "Vision Prism source provenance rejected: ${VERIFY_ERROR}")
endif()
string(JSON VISION_ABI GET "${PROVENANCE}" visionABI)
string(JSON VISION_REVISION GET "${PROVENANCE}" revision)
string(JSON VISION_PATCH_SHA256 GET "${PROVENANCE}" visionPatchSha256)
if(NOT VISION_ABI EQUAL 1)
  message(FATAL_ERROR "Vision source ABI 1 is required")
endif()
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
if(NOT TARGET llama)
  if(TARGET ggml OR TARGET mtmd)
    message(FATAL_ERROR "Conflicting runtime targets exist before llama")
  endif()
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${PRISM_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/prism-upstream" EXCLUDE_FROM_ALL)
elseif(NOT TARGET mtmd)
  # This includes the library target only; never configure a second runtime.
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${PRISM_SOURCE_DIR}/tools/mtmd" "${CMAKE_BINARY_DIR}/quenderin-mtmd" EXCLUDE_FROM_ALL)
endif()
add_library(quenderin_vision STATIC quenderin_vision.cpp vision_safety.cpp quenderin_vision.h)
target_include_directories(quenderin_vision PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${PRISM_SOURCE_DIR}/vendor")
target_compile_definitions(quenderin_vision PRIVATE QUENDERIN_VISION_BUILD_DLL
  QUENDERIN_VISION_REVISION="${VISION_REVISION}" QUENDERIN_VISION_PATCH_SHA256="${VISION_PATCH_SHA256}")
target_link_libraries(quenderin_vision PUBLIC llama mtmd PRIVATE vendor::hash)
set_target_properties(quenderin_vision PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(QUENDERIN_VISION_BUILD_CLI "Build bounded desktop vision worker" ON)
if(QUENDERIN_VISION_BUILD_CLI AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(quenderin-vision-cli vision_cli.cpp)
  target_include_directories(quenderin-vision-cli PRIVATE "${PRISM_SOURCE_DIR}/vendor")
  target_compile_definitions(quenderin-vision-cli PRIVATE QUENDERIN_VISION_BUILD_DLL)
  target_link_libraries(quenderin-vision-cli PRIVATE quenderin_vision)
  set_target_properties(quenderin-vision-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(quenderin-vision-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  if(MSVC)
    target_compile_options(quenderin_vision PRIVATE /utf-8 /EHsc)
    target_compile_options(quenderin-vision-cli PRIVATE /utf-8 /EHsc)
  elseif(UNIX AND NOT APPLE)
    target_link_options(quenderin-vision-cli PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
