cmake_minimum_required(VERSION 3.21)
project(quenderin_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
set(TRANSCRIBE_SOURCE "" CACHE PATH "Path to verified pinned anomalyco/transcribe.cpp source")
if(NOT TRANSCRIBE_SOURCE)
  message(FATAL_ERROR "Set TRANSCRIBE_SOURCE to the verified pinned transcribe.cpp checkout")
endif()
if(NOT EXISTS "${TRANSCRIBE_SOURCE}/include/transcribe.h")
  message(FATAL_ERROR "Missing ${TRANSCRIBE_SOURCE}/include/transcribe.h")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../scripts/prepare_speech_runtime.py" verify --source "${TRANSCRIBE_SOURCE}"
  RESULT_VARIABLE SPEECH_SOURCE_OK OUTPUT_VARIABLE SPEECH_SOURCE_OUT ERROR_VARIABLE SPEECH_SOURCE_ERR)
if(NOT SPEECH_SOURCE_OK EQUAL 0)
  message(FATAL_ERROR "Pinned speech source verification failed: ${SPEECH_SOURCE_ERR}")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../shared/speech-runtime.json" SPEECH_RUNTIME_JSON)
string(JSON SPEECH_REVISION GET "${SPEECH_RUNTIME_JSON}" "revision")

set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
if(APPLE)
  # ObjC runtime class names are global even when Mach-O exports are hidden.
  add_compile_definitions(GGMLMetalClass=QuenderinSpeechMetalClass)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${TRANSCRIBE_SOURCE}" transcribe-cpp EXCLUDE_FROM_ALL)
# Upstream transcribe.cpp/src/CMakeLists.txt specifies $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/include>,
# which resolves to native/speech instead of TRANSCRIBE_SOURCE when added via add_subdirectory.
target_include_directories(transcribe PUBLIC "${TRANSCRIBE_SOURCE}/include")

add_library(quenderin_speech SHARED quenderin_speech.cpp quenderin_speech.h)
target_compile_definitions(quenderin_speech PRIVATE QUENDERIN_SPEECH_BUILD_DLL QUENDERIN_SPEECH_REVISION="${SPEECH_REVISION}")
target_include_directories(quenderin_speech PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${TRANSCRIBE_SOURCE}/include")
target_link_libraries(quenderin_speech PRIVATE transcribe)
set_target_properties(quenderin_speech PROPERTIES CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set_target_properties(quenderin_speech PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  target_sources(quenderin_speech PRIVATE exports.def)
  set_target_properties(quenderin_speech PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF PREFIX "" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(quenderin_speech PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin"
      ARCHIVE_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/lib")
  endforeach()
  if(MSVC)
    target_compile_options(quenderin_speech PRIVATE /utf-8 /EHs)
  endif()
elseif(APPLE)
  include("${CMAKE_CURRENT_SOURCE_DIR}/AppleFrameworkMetadata.cmake")
  target_sources(quenderin_speech PRIVATE module.modulemap)
  set_source_files_properties(module.modulemap PROPERTIES MACOSX_PACKAGE_LOCATION Modules)
  set_target_properties(quenderin_speech PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_FRAMEWORK_IDENTIFIER org.quenderin.speech PUBLIC_HEADER quenderin_speech.h
    MACOSX_FRAMEWORK_BUNDLE_NAME QuenderinSpeech
    MACOSX_FRAMEWORK_BUNDLE_VERSION 1 MACOSX_FRAMEWORK_SHORT_VERSION_STRING 1.0.0
    VERSION 1.0.0 SOVERSION 1 OUTPUT_NAME QuenderinSpeech INSTALL_NAME_DIR "@rpath")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_custom_command(TARGET quenderin_speech POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E create_symlink "Versions/Current/Modules"
        "$<TARGET_BUNDLE_DIR:quenderin_speech>/Modules"
      VERBATIM)
  endif()
  target_link_options(quenderin_speech PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.apple")
elseif(UNIX)
  target_link_options(quenderin_speech PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic")
  if(ANDROID)
    target_link_options(quenderin_speech PRIVATE "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
    add_library(quenderin_speech_jni SHARED speech_jni.cpp)
    target_include_directories(quenderin_speech_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(quenderin_speech_jni PRIVATE quenderin_speech)
    set_target_properties(quenderin_speech_jni PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    target_link_options(quenderin_speech_jni PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.jni.map" "LINKER:--exclude-libs,ALL" "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
  endif()
else()
  message(FATAL_ERROR "This isolated native build supports Windows, Apple and ELF platforms only.")
endif()

option(QUENDERIN_SPEECH_BUILD_CLI "Build local desktop speech CLI adapter" ON)
if(QUENDERIN_SPEECH_BUILD_CLI AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(quenderin-speech-cli speech_cli.cpp)
  target_link_libraries(quenderin-speech-cli PRIVATE quenderin_speech)
  if(APPLE)
    set_target_properties(quenderin-speech-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" BUILD_RPATH "@executable_path")
  else()
    set_target_properties(quenderin-speech-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" BUILD_RPATH "$ORIGIN")
  endif()
  if(WIN32)
    foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
      set_target_properties(quenderin-speech-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
    endforeach()
    if(MSVC)
      target_compile_options(quenderin-speech-cli PRIVATE /utf-8 /EHs)
    elseif(MINGW)
      target_link_options(quenderin-speech-cli PRIVATE -municode)
    endif()
  endif()
endif()
