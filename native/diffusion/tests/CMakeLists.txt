# Hosted Windows filesystem verification; synthetic SD API, no model weights.
cmake_minimum_required(VERSION 3.22)
project(quenderin_diffusion_windows_fixtures LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Use scripts/test_diffusion_core.py for non-Windows host fixtures.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
add_library(quenderin_diffusion SHARED ../quenderin_diffusion.cpp stable-diffusion.cpp ../exports.def)
target_include_directories(quenderin_diffusion PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(quenderin_diffusion PRIVATE QUENDERIN_DIFFUSION_BUILD_DLL QUENDERIN_DIFFUSION_REVISION="synthetic-windows-fixture")
set_target_properties(quenderin_diffusion PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF PREFIX "")
add_executable(quenderin-diffusion-windows-fixture windows_filesystem.cpp)
target_link_libraries(quenderin-diffusion-windows-fixture PRIVATE quenderin_diffusion)
add_executable(quenderin-diffusion-cli ../diffusion_cli.cpp)
target_link_libraries(quenderin-diffusion-cli PRIVATE quenderin_diffusion)
foreach(target quenderin_diffusion quenderin-diffusion-windows-fixture quenderin-diffusion-cli)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /EHsc)
  elseif(MINGW AND target STREQUAL "quenderin-diffusion-cli")
    target_link_options(${target} PRIVATE -municode)
  endif()
endforeach()
enable_testing()
add_test(NAME windows-exclusive-png COMMAND quenderin-diffusion-windows-fixture)
add_test(NAME windows-reparse-rejection COMMAND quenderin-diffusion-windows-fixture reparse)
set_tests_properties(windows-reparse-rejection PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME windows-short-path-alias-rejection COMMAND quenderin-diffusion-windows-fixture short-alias)
set_tests_properties(windows-short-path-alias-rejection PROPERTIES SKIP_RETURN_CODE 77)
add_test(NAME windows-cli-entry-version COMMAND quenderin-diffusion-cli --version)
set_tests_properties(windows-cli-entry-version PROPERTIES PASS_REGULAR_EXPRESSION "abiVersion.*1")
