cmake_minimum_required(VERSION 3.22)
project(quenderin_diffusion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS OFF)
if(TARGET ggml OR TARGET llama OR TARGET stable-diffusion)
  message(FATAL_ERROR "Diffusion must use its own CMake configure: its patched GGML ABI is isolated from llama.")
endif()
if(NOT DEFINED DIFFUSION_SOURCE_DIR)
  message(FATAL_ERROR "Set DIFFUSION_SOURCE_DIR to the verified pinned source checkout.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/prepare_diffusion_runtime.py"
  verify --source "${DIFFUSION_SOURCE_DIR}" RESULT_VARIABLE VERIFIED OUTPUT_VARIABLE PROVENANCE ERROR_VARIABLE VERIFY_ERROR)
if(NOT VERIFIED EQUAL 0)
  message(FATAL_ERROR "Diffusion source provenance rejected: ${VERIFY_ERROR}")
endif()
string(JSON DIFFUSION_REVISION GET "${PROVENANCE}" revision)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_USE_SYSTEM_GGML OFF CACHE BOOL "" FORCE)
set(SD_USE_UPSTREAM_GGML OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(APPLE)
  # ObjC runtime class names are global even when Mach-O exports are hidden.
  # Prefix the pinned Metal backend's sole ObjC helper class as well.
  add_compile_definitions(GGMLMetalClass=QuenderinDiffusionMetalClass)
  set(SD_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${DIFFUSION_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/diffusion-upstream" EXCLUDE_FROM_ALL)
add_library(quenderin_diffusion SHARED quenderin_diffusion.cpp)
target_compile_definitions(quenderin_diffusion PRIVATE QUENDERIN_DIFFUSION_BUILD_DLL QUENDERIN_DIFFUSION_REVISION="${DIFFUSION_REVISION}")
target_include_directories(quenderin_diffusion PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${DIFFUSION_SOURCE_DIR}/include")
target_link_libraries(quenderin_diffusion PRIVATE stable-diffusion)
set_target_properties(quenderin_diffusion PROPERTIES CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set_target_properties(quenderin_diffusion PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(WIN32)
  target_sources(quenderin_diffusion PRIVATE exports.def)
  set_target_properties(quenderin_diffusion PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF PREFIX ""
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
  foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(quenderin_diffusion PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin"
      ARCHIVE_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/lib")
  endforeach()
  if(MSVC)
    target_compile_options(quenderin_diffusion PRIVATE /utf-8 /EHsc)
  endif()
elseif(APPLE)
  target_sources(quenderin_diffusion PRIVATE module.modulemap)
  set_source_files_properties(module.modulemap PROPERTIES MACOSX_PACKAGE_LOCATION Modules)
  set_target_properties(quenderin_diffusion PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER org.quenderin.diffusion PUBLIC_HEADER quenderin_diffusion.h
    OUTPUT_NAME QuenderinDiffusion INSTALL_NAME_DIR "@rpath")
  target_link_options(quenderin_diffusion PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.apple")
elseif(UNIX)
  target_link_options(quenderin_diffusion PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic")
  if(ANDROID)
    target_link_options(quenderin_diffusion PRIVATE "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
    add_library(quenderin_diffusion_jni SHARED diffusion_jni.cpp)
    target_include_directories(quenderin_diffusion_jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(quenderin_diffusion_jni PRIVATE quenderin_diffusion)
    set_target_properties(quenderin_diffusion_jni PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    target_link_options(quenderin_diffusion_jni PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.jni.map" "LINKER:--exclude-libs,ALL" "LINKER:-z,max-page-size=16384" "LINKER:-z,common-page-size=16384")
  endif()
else()
  message(FATAL_ERROR "This isolated native build supports Windows, Apple and ELF platforms only.")
endif()
option(QUENDERIN_DIFFUSION_BUILD_CLI "Build local desktop image CLI adapter" ON)
if(QUENDERIN_DIFFUSION_BUILD_CLI AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(quenderin-diffusion-cli diffusion_cli.cpp)
  target_link_libraries(quenderin-diffusion-cli PRIVATE quenderin_diffusion)
  set_target_properties(quenderin-diffusion-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" BUILD_RPATH "$ORIGIN;@executable_path")
  if(WIN32)
    foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
      set_target_properties(quenderin-diffusion-cli PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
    endforeach()
    if(MSVC)
      target_compile_options(quenderin-diffusion-cli PRIVATE /utf-8 /EHsc)
    elseif(MINGW)
      target_link_options(quenderin-diffusion-cli PRIVATE -municode)
    endif()
  endif()
endif()
